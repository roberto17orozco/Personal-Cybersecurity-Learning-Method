 23 # ANSI Colors
 24 BLUE="\e[34m"

 30 # Logging utilities

 32 log_warn()      { echo "[!] $*"; }
 33 log_error()     { echo "[-] $*" >&2; }
 34 
 35 # Status messages (blue)
 36 status () { echo -e "${BLUE}[STATUS] $*${RESET}"; }

 38 ensure_report_dir() {

 42 ################################################
 43 # 3. CORE MODULE - SYSTEM HARDENING ENGINE
 44 ################################################
 45 # This is the dominant conceptual module.

 47 #
 48 run_hardening_checks() {
 49     status "Running core hardening checks..."
 50 
 51     set +e
 52     {
 53         echo "========================================"
 54         echo " System Hardening Report"
 55         echo " Version: $VERSION"
 56         echo " Date: $(date)"
 57         echo "========================================"
 58         echo
 59         echo "1. File Permissions"

 61         check_critical_permissions

 63         echo "2. Running Services"
 64         echo "----------------------"

 67         echo "3. World-Writable Files"

 70         echo
 71         echo "4. SUID/SGID Binaries"
 72         echo "-----------------------"
 73         check_suid_sgid

 75         echo "5. Firewall Status"
 76         echo "-----------------------"
 77         check_firewall
 78         echo
 79         echo "6. SSH Configuration"
 80         echo "-----------------------"
 81         check_ssh_config
 82         echo
 83     } > "$REPORT_FILE"
 84     set -e
 85 
 86     log_info "Report generated: $REPORT_FILE"
 87 }

 89 #############################################

 93 check_critical_permissions() {
 94     # Critical system files and their expected permissions
 95     declare -A expected_perms=(
 96         ["/etc/passwd"]=644
 97         ["/etc/group"]=644
 98         ["/etc/shadow"]=640
 99         ["/etc/gshadow"]=640
100         ["/etc/sudoers"]=440
101     )
102 
103 
104 
105     for f in "${!expected_perms[@]}"; do
106         if [[ -e "$f" ]]; then
107             perms=$(stat -c "%a" "$f")
108             expected="${expected_perms[$f]}"
109 
110             echo "$f permissions: $perms"
111 
112             # Special rule: passwd and group are allowed to be 644
113             if [[ "$f" == "/etc/passwd" || "$f" == "/etc/group" ]]; then
114                 if [[ "$perms" -ne 644 ]]; then
115                     log_warn "$f has non-standard permissions ($perms), expected 644"
116                 else
117                     log_info "$f permissions are secure ($perms)"
118                 fi
119                 continue
120             fi
121 
122             # For shadow, gshadow, sudoers
123             if [[ "$perms" -ne "$expected" ]]; then
124                 log_warn "$f has weak permissions ($perms), expected $expected"
125             else
126                 log_info "$f permissions are secure ($perms)"
127             fi
128         else
129             log_error "$f not found"
130         fi
131     done
132 }
133 
134 
135 check_running_services() {
136     local suspicious=("telnet" "vsftpd" "rsh" "rexec")
137 
138     for svc in "${suspicious[@]}"; do
139         if systemctl is-active --quiet "$svc"; then
140             log_warn "Suspicious service running: $svc"
141         else
142             log_info "Service not running: $svc"
143         fi
144     done
145 }
146 
147 
148 check_world_writable() {
149     local count
150     count=$(find / -xdev -type f -perm -0002 2>/dev/null | wc -l)
151     echo "World-writable files: $count"
152 
153     if [[ "$count" -gt 50 ]]; then
154         log_warn "High number of world-writable files"
155     else
156         log_info "World-writable files count acceptable"
157     fi
158 }
159 
160 check_suid_sgid() {
161     local count
162     count=$(find / -xdev \( -perm -4000 -o -perm -2000 \) 2>/dev/null | wc -l)
163 
164     echo "SUID/SGID binaries: $count"
165 
166     if [[ "$count" -gt 100 ]]; then
167         log_warn "High number of SUID/SGID binaries"
168     else
169         log_info "SUID/SGID count acceptable"
170     fi
171 }
172 
173 check_firewall() {
174     if command -v ufw >/dev/null 2>&1; then
175         ufw status | sed 's/^/    /'
176     else
177         log_warn "UFW not installed"
178     fi
179 }
180 
181 check_ssh_config() {
182     if [[ -f /etc/ssh/sshd_config ]]; then
183         if grep -q "^PermitRootLogin yes" /etc/ssh/sshd_config; then
184             log_warn "Root login over SSH is enabled"
185         else
186             log_info "Root login over SSH is disabled"
187         fi
188     else
189         log_error "SSH configuration file not found"
190     fi
191 }
192

194 # 5. ORCHESTRATOR MODULE
195 #######################################

198     status "Initializing report directory..."
199     ensure_report_dir

201     status "Starting hardening evaluation..."

203 
204     status "Process completed."
205 }
