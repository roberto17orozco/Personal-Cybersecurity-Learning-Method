1 #!/usr/bin/env bash
  2 
  3 # ============================
  4 #   ANSI COLORS
  5 # ============================
  6 RED="\e[31m"
  7 GREEN="\e[32m"
  8 YELLOW="\e[33m"
  9 BLUE="\e[34m"
 10 CYAN="\e[36m"
 11 RESET="\e[0m"
 12 
 13 TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
 14 OUTPUT="traffic_report_$TIMESTAMP.txt"
 15 MIN_PORTS=10
 16 
 17 echo -e "${CYAN}[INFO] Starting PRO++ traffic detector...${RESET}"
 18 
 19 # ============================
 20 #   LOCAL NETWORK INFORMATION
 21 # ============================
 22 echo -e "${BLUE}[INFO] Gathering local network information...${RESET}"
 23 
 24 IFACE=$(ip route | awk '/default/ {print $5}')
 25 LOCAL_IP=$(ip -4 addr show "$IFACE" | awk '/inet/ {print $2}' | cut -d'/' -f1)
 26 LOCAL_MASK=$(ip -4 addr show "$IFACE" | awk '/inet/ {print $2}' | cut -d'/' -f2)
 27 LOCAL_NET=$(ipcalc "$LOCAL_IP/$LOCAL_MASK" | awk -F': *' '/Network/ {print $2}')
 28 LOCAL_GW=$(ip route | awk '/default/ {print $3}')
 29 
 30 print_banner() {
 31     echo "=============================================="
 32     echo "     TRAFFIC & PORT SCAN DETECTOR (PRO++)     "
 33     echo "=============================================="
 34     echo
 35     echo "Local network information (victim - Kali):"
 36     echo "  Interface: $IFACE"
 37     echo "  Local IP: $LOCAL_IP"
 38     echo "  Netmask: /$LOCAL_MASK"
 39     echo "  Network: $LOCAL_NET"
 40     echo "  Gateway: $LOCAL_GW"
 41     echo
 42 }
 43 
 44 # ============================
 45 #   ATTACK DETECTION
 46 # ============================
 47 detect_scans() {
 48 
 49     echo -e "${YELLOW}[INFO] Analyzing firewall logs...${RESET}"
 50 
 51     LOG_DATA=$(sudo journalctl -g "IPTABLES DROP" -o short)
 52 
 53     if [[ -z "$LOG_DATA" ]]; then
 54         echo "No firewall events detected." > "$OUTPUT"
 55         return
 56     fi
 57 
 58     echo "$LOG_DATA" | awk -v min="$MIN_PORTS" -v mask="$LOCAL_MASK" -v net="$LOCAL_NET" -v gw="$LOCAL_GW" '
 59 
 60         function is_private(ip) {
 61             split(ip, o, ".")
 62             return ((o[1] == 10) || (o[1] == 172 && o[2] >= 16 && o[2] <= 31) || (o[1] == 192 && o[2] == 168))
 63         }
 64 
 65         {
 66             ts = $1 " " $2 " " $3
 67             src = ""
 68             dpt = ""
 69             proto = ""
 70             icmp = 0
 71             syn = 0
 72 
 73             for (i=1; i<=NF; i++) {
 74                 if ($i ~ /^SRC=/) { split($i,a,"="); src=a[2] }
 75                 if ($i ~ /^DPT=/) { split($i,b,"="); dpt=b[2] }
 76                 if ($i ~ /^PROTO=/) { split($i,c,"="); proto=c[2] }
 77                 if ($i ~ /^TYPE=/) { icmp=1 }
 78                 if ($i ~ /SYN/) { syn=1 }
 79             }
 80 
 81             if (src != "") {
 82                 if (dpt != "") ports[src][dpt] = 1
 83 
 84                 if (!(src in first)) first[src] = ts
 85                 last[src] = ts
 86 
 87                 if (proto != "") protos[src][proto]++
 88                 if (icmp == 1) icmp_count[src]++
 89                 if (syn == 1) syn_count[src]++
 90 
 91                 if (dpt == 22) ssh_count[src]++
 92                 if (dpt == 80 || dpt == 8080) http_count[src]++
 93                 if (proto == "UDP") udp_count[src]++
 94             }
 95         }
 96 
 97         END {
 98             for (ip in first) {
 99                 printf "Attacker IP: %s\n", ip
100 
101                 if (is_private(ip))
102                     printf "  IP type: Private (RFC1918)\n"
103                 else
104                     printf "  IP type: Public\n"
105 
106                 if (is_private(ip)) {
107                     printf "  Victim netmask: /%s\n", mask
108                     printf "  Network (CIDR): %s\n", net
109                     printf "  Local gateway: %s\n", gw
110                 }
111 
112                 port_count = 0
113                 minp = 99999
114                 maxp = 0
115 
116                 for (p in ports[ip]) {
117                     port_count++
118                     if (p < minp) minp = p
119                     if (p > maxp) maxp = p
120                 }
121 
122                 if (port_count > 0) {
123                     printf "  Distinct ports: %d\n", port_count
124                     printf "  Port range: %d - %d\n", minp, maxp
125                 }
126 
127                 printf "  First event: %s\n", first[ip]
128                 printf "  Last event: %s\n", last[ip]
129 
130                 if (port_count >= min) {
131                     if (maxp - minp <= port_count + 5)
132                         printf "  Scan type: Sequential\n"
133                     else
134                         printf "  Scan type: Random\n"
135                 }
136 
137                 printf "  Traffic nature:\n"
138 
139                 if (icmp_count[ip] > 0)
140                     printf "    - ICMP (possible ping)\n"
141 
142                 if (ssh_count[ip] > 0)
143                     printf "    - SSH (port 22 attempts)\n"
144 
145                 if (http_count[ip] > 0)
146                     printf "    - HTTP (curl or browser)\n"
147 
148                 if (udp_count[ip] > 0)
149                     printf "    - UDP traffic\n"
150 
151                 if (syn_count[ip] > 0)
152                     printf "    - SYN-only packets (possible Nmap scan)\n"
153 
154                 if (port_count >= min)
155                     printf "    - Port scanning (possible nmap)\n"
156 
157                 if (icmp_count[ip] == 0 && ssh_count[ip] == 0 && http_count[ip] == 0 && udp_count[ip] == 0 && port_count < min)
158                     printf "    - Generic TCP traffic\n"
159 
160                 printf "-----------------------------\n"
161             }
162         }
163     ' > "$OUTPUT"
164 }
165 
166 main() {
167     print_banner
168     detect_scans
169     echo -e "${GREEN}[OK] Report generated: ${OUTPUT}${RESET}"
170 }
171 
172 main
