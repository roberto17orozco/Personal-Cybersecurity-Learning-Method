 33 log_info()  { echo -e "${GREEN}[+]${RESET} $*"; }

 35 log_error() { echo -e "${RED}[-]${RESET} $*" >&2; }

 80 ###############################################
 81 # 4. SUPORTING EVALUATORS (Journalctl-based)
 82 ###############################################
 83 
 84 correlate_failed_ssh() {
 85     journalctl -u ssh --no-pager --since "24 hours ago" \
 86         | grep "Failed password" \
 87         | awk '{print $1, $2, $3, $11}' \
 88         || log_warn "No failed SSH attempts found"
 89 }
 90 
 91 correlate_root_activity() {
 92     journalctl -u ssh --no-pager --since "24 hours ago" \
 93         | grep "session opened for user root" \
 94         | awk '{print $1, $2, $3, $11}' \
 95         || log_warn "No root SSH sessions found"
 96 }
 97 
 98 correlate_kernel_security() {
 99     journalctl -k --no-pager --since "24 hours ago" \
100         | grep -Ei "audit|apparmor|denied|security" \
101         || log_warn "No kernel security events found"
102 }
103 
104 correlate_multi_source() {
105     log_info "Correlating SSH failures with sudo events (24h window)..."
106 
107     # ------------------------------
108     # 1. Get SSH failures (only once)
109     # ------------------------------
110     mapfile -t ssh_failures < <(
111         journalctl -u ssh --no-pager --since "24 hours ago" \
112             | grep "Failed password" \
113             | awk '{print $1" "$2" "$3, $11}'
114         )
115 
116     # ------------------------------
117     # 2. Get sudo events (only once)
118     # ------------------------------
119     mapfile -t sudo_events < <(
120         journalctl -t sudo --no-pager --since "24 hours ago"
121     )
122 
123     # If no events, exit
124     [[ ${#ssh_failures[@]} -eq 0 ]] && {
125         log_warn "No SSH failures detected"
126         return
127     }
128 
129     [[ ${#sudo_events[@]} -eq 0 ]] && {
130         log_warn "No sudo events detected"
131         return
132     }
133 
134     # ------------------------------
135     # 3. Actual correlation
136     # ------------------------------
137     local correlated=0
138 
139     for entry in "${ssh_failures[@]}"; do
140         ts=$(awk '{print $1" "$2" "$3}' <<< "$entry")
141         ip=$(awk '{print $4}' <<< "$entry")
142 
143         epoch=$(date -d "$ts" +"%s" 2>/dev/null || echo 0)
144 
145         for line in "${sudo_events[@]}"; do
146             log_ts=$(awk '{print $1" "$2" "$3}' <<< "$line")
147             log_epoch=$(date -d "$log_ts" +"%s" 2>/dev/null || echo 0)
148 
149             # Correlation window: 60 seconds
150             if (( log_epoch >= epoch && log_epoch <= epoch+60 )); then
151                 echo "[CORRELATED] SSH failure from $ip followed by sudo event: $line"
152                 correlated=1
153             fi
154         done
155     done
156 
157     # -----------------------------
158     # 4. Final result
159     # -----------------------------
160     (( correlated == 0 )) && log_warn "No multi-source correlations detected"
161 }
