  9 # Colors
 10 RED="\e[31m"
 11 GREEN="\e[32m"
 12 YELLOW="\e[33m"
 13 BLUE="\e[34m"
 14 RESET="\e[0m"
 15 
 16 info()  { echo -e "${BLUE}[INFO]${RESET} $1" >&2; }
 17 ok()    { echo -e "${GREEN}[+]${RESET} $1" >&2; }
 18 warn()  { echo -e "${YELLOW}[WARN]${RESET} $1" >&2; }
 19 
 20 TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
 21 REPORT="process_monitor_report_$TIMESTAMP.txt"
 22 
 23 # ============================================================
 24 #  CONFIG
 25 # ============================================================
 26 CPU_THRESHOLD=25
 27 SUSPICIOUS_DIRS=("/tmp" "/var/tmp" "/dev/shm" "/home/$USER/Downloads")
 28 WATCH_INTERVAL=5
 29 
 30 # ============================================================
 31 #  HELP
 32 # ============================================================
 33 show_help() {
 34     echo "Usage: $0 [--cpu | --network | --paths | --full | --watch N]"
 35     exit 0
 36 }
 37 
 38 # ============================================================
 39 #  UTILS
 40 # ============================================================
 41 log() {
 42     echo "$1" >> "$REPORT"
 43 }
 44 
 45 is_suspicious_path() {
 46     for dir in "${SUSPICIOUS_DIRS[@]}"; do
 47         [[ "$1" == $dir* ]] && return 0
 48     done
 49     return 1
 50 }
 51 
 52 # ============================================================
 53 #  MODULE 1 – CPU ANOMALIES
 54 # ============================================================
 55 check_cpu_anomaly() {
 56     info "Checking for high CPU usage processes..."
 57     log "=== High CPU Usage Processes (> ${CPU_THRESHOLD}%) ==="
 58     ps aux --sort=-%cpu | awk -v th="$CPU_THRESHOLD" '
 59         $3 > th {
 60             printf "PID: %s | CPU: %s%% | USER: %s | CMD: %s\n", $2, $3, $1, $11
 61         }
 62     ' >> "$REPORT"
 63     ok "CPU anomaly check completed."
 64     log ""
 65 }
 66 
 67 # ============================================================
 68 #  MODULE 2 – SUSPICIOUS EXECUTION PATHS
 69 # ============================================================
 70 check_suspicious_paths() {
 71     info "Scanning for processes running from suspicious directories..."
 72     log "=== Processes Running from Suspicious Directories ==="
 73     for pid in $(ls /proc | grep -E '^[0-9]+$'); do
 74         exe=$(readlink -f /proc/$pid/exe 2>/dev/null)
 75         [[ -z "$exe" ]] && continue
 76 
 77         if is_suspicious_path "$exe"; then
 78             cmd=$(tr -d '\0' < /proc/$pid/cmdline)
 79             log "PID: $pid | EXE: $exe | CMD: $cmd"
 80         fi
 81     done
 82     ok "Suspicious path scan completed."
 83     log ""
 84 }
 85 
 86 # ============================================================
 87 #  MODULE 3 – NETWORK-ACTIVE PROCESSES
 88 # ============================================================
 89 check_network_activity() {
 90     info "Checking for processes with open network ports..."
 91     log "=== Processes with Open Network Ports ==="
 92     ss -tulnp 2>/dev/null | awk '
 93         NR>1 {
 94             printf "Protocol: %s | Local: %s | PID/Program: %s\n", $1, $5, $7
 95         }
 96     ' >> "$REPORT"
 97     ok "Network activity scan completed."
 98     log ""
 99 }
100 
101 # ============================================================
102 #  MODULE 4 – SHORT-LIVED PROCESSES
103 # ============================================================
104 check_short_lived() {
105     info "Detecting short-lived processes..."
106     log "=== Short-Lived Processes (appear/disappear quickly) ==="
107     declare -A seen
108 
109     for i in {1..2}; do
110         for pid in $(ps -e -o pid=); do
111             seen[$pid]=$((seen[$pid]+1))
112         done
113         sleep 1
114     done
115 
116     for pid in "${!seen[@]}"; do
117         if [[ ${seen[$pid]} -eq 1 ]]; then
118             cmd=$(ps -p $pid -o cmd= 2>/dev/null)
119             log "PID: $pid | CMD: $cmd"
120         fi
121     done
122     ok "Short-lived process detection completed."
123     log ""
124 }
125 
126 # ============================================================
127 #  MODULE 5 – ROOT PROCESSES (Unexpected)
128 # ============================================================
129 check_root_processes() {
130     info "Checking for unexpected root processes..."
131     log "=== Unexpected Root Processes ==="
132     ps -U root -u root u | awk '
133         $11 !~ /(systemd|NetworkManager|dbus-daemon|cron|rsyslogd|sshd)/ {
134             printf "PID: %s | USER: %s | CMD: %s\n", $2, $1, $11
135         }
136     ' >> "$REPORT"
137     ok "Root process scan completed."
138     log ""
139 }
140 
141 # ============================================================
142 #  FULL SCAN
143 # ============================================================
144 run_full_scan() {
145     info "Generating full report: $REPORT"
146     log "====================================================="
147     log " Suspicious Process Monitor PRO – Full Scan"
148     log " Generated: $(date)"
149     log "====================================================="
150     log ""
151 
152     check_cpu_anomaly
153     check_suspicious_paths
154     check_network_activity
155     check_short_lived
156     check_root_processes
157 
158     ok "Full scan completed."
159     echo -e "${GREEN}[+]${RESET} Report saved to: $REPORT" >&2
160 }
161 
162 # ============================================================
163 #  WATCH MODE
164 # ============================================================
165 run_watch_mode() {
166     interval=$1
167     info "Running continuous monitoring every $interval seconds..."
168     echo "Press CTRL+C to stop." >&2
169 
170     while true; do
171         REPORT="watch_report_$(date +"%H-%M-%S").txt"
172         run_full_scan >/dev/null
173         sleep "$interval"
174     done
175 }
176 
177 # ============================================================
178 #  CLI HANDLER
179 # ============================================================
180 case "$1" in
181     --cpu) check_cpu_anomaly ;;
182     --paths) check_suspicious_paths ;;
183     --network) check_network_activity ;;
184     --full) run_full_scan ;;
185     --watch) run_watch_mode "$2" ;;
186     *) show_help ;;
187 esac
188 
189 # ============================================================
190 #  END OF SCRIPT
191 # ============================================================
192 echo -e "${BLUE}[INFO]${RESET} Suspicious Process Monitor PRO finished execution." >&2
