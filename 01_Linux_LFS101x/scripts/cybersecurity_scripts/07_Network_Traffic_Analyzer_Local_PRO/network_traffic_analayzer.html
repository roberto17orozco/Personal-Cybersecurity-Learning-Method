 14 #############################################
 15 # 1. GLOBAL CONFIG & CORE STATE (Core-First)
 16 #############################################
 17 
 18 VERSION="1.0.0"
 19 DEFAULT_IFACE="eth0"
 20 DEFAULT_CAPTURE_DURATION=20
 21 DEFAULT_WATCH_INTERVAL=60
 22 DEFAULT_CAPTURE_DIR="$HOME/network_captures"
 23 MIN_SUSPICIOUS_PORTS=10
 24 
 25 CAPTURE_TOOL=""
 26 
 27 
 28 #############################################
 29 # 2. UTILITY LAYER (Cross-cutting utilities)
 30 #############################################
 31 
 32 
 33 # Color definitions for status messages
 34 BLUE="\e[34m"
 35 RESET="\e[0m"
 36 
 37 log_info() { echo "[+] $*"; }
 38 log_warn() { echo "[!] $*" >&2; }
 39 log_error() { echo "[-] $*" >&2; }
 40 
 41 timestamp () { date +"%Y-%m-%d_%H-%M-%S"; }
 42 
 43 require_cmd() {
 44     local cmd="$1"
 45     command -v "$cmd" >/dev/null 2>&1
 46 }
 47 
 48 init_capture_dir() {
 49     local dir="$1"
 50     if [[ ! -d "$dir" ]]; then
 51         mkdir -p "$dir"
 52         log_info "Capture directory created: $dir"
 53     fi
 54 }
 55 
 56 detect_capture_tool() {
 57     if require_cmd tshark; then
 58         CAPTURE_TOOL="tshark"
 59         log_info "Using tshark as capture tool."
 60     elif require_cmd tcpdump; then
 61         CAPTURE_TOOL="tcpdump"
 62         log_info "Using tcpdump as capture tool."
 63     else
 64         log_error "Neither tshark nor tcpdump is installed."
 65         exit 1
 66     fi
 67 }
 68 
 69 
 70 #######################################
 71 # 3. CORE CAPTURE ENGINE
 72 #######################################
 73 
 74 run_capture_once() {
 75     local iface="$1"
 76     local duration="$2"
 77     local out_dir="$3"
 78 
 79 
 80     local ts
 81     ts="$(timestamp)"
 82     local pcap_file="${out_dir}/capture_${ts}.pcap"
 83 
 84     log_info "Starting packet capture on ${iface} for ${duration}s..." >&2
 85 
 86     case "$CAPTURE_TOOL" in
 87         tshark)
 88             tshark -i "$iface" -a "duration:${duration}" -w "$pcap_file"
 89             ;;
 90         tcpdump)
 91             timeout --signal=INT "${duration}" tcpdump -i "$iface" -w "$pcap_file" > /dev/null 2>&1 || true
 92             ;;
 93     esac
 94 
 95     if [[ -f "$pcap_file" ]]; then
 96         log_info "Capture completed: $pcap_file" >&2
 97     else
 98         log_warn "Capture file was not created." >&2
 99     fi
100 
101     echo "$pcap_file"
102 }
103 
104 ######################################
105 # 4. CORE ANALYSIS PRIMITIVES
106 ######################################
107 
108 analyze_protocols() {
109     local pcap="$1"
110     [[ -f "$pcap" ]] || { echo "No pcap file available."; return; }
111 
112     if [[ "$CAPTURE_TOOL" == "tshark" ]]; then
113         tshark -r "$pcap" -q -z io,phs 2>/dev/null
114     else
115         tcpdump -nn -r "$pcap" 2>/dev/null \
116             | awk '{print $5}' | sed 's/://g' \
117             | awk -F'.' '{print $NF}' \
118             | sort | uniq -c | sort -nr
119     fi
120 }
121 
122 analyze_endpoints() {
123     local pcap="$1"
124     [[ -f "$pcap" ]] || { echo "No pcap file available."; return; }
125 
126     if [[ "$CAPTURE_TOOL" == "tshark" ]]; then
127         tshark -r "$pcap" -q -z endpoints,ip 2>/dev/null
128     else
129         tcpdump -nn -r "$pcap" 2>/dev/null \
130             | awk '{print $3, $5}' | sed 's/://g' \
131             | sort | uniq -c | sort -nr | head -50
132     fi
133 }
134 
135 analyze_connections() {
136     local pcap="$1"
137     [[ -f "$pcap" ]] || { echo "No pcap file available."; return; }
138 
139     if [[ "$CAPTURE_TOOL" == "tshark" ]]; then
140         tshark -r "$pcap" -T fields \
141             -e ip.src -e tcp.srcport -e ip.dst -e tcp.dstport -e tcp.flags\
142             -Y "tcp" 2>/dev/null \
143             | awk 'NF==5 {printf "%-15s:%-5s -> %-15s:%-5s flags=%s\n", $1,$2,$3,$4,$5}' \
144             | sort | uniq -c | sort -nr | head -50
145     else
146         tcpdump -nn -r "$pcap" tcp 2>/dev/null \
147             | awk '{print $3, $5}' | sed 's/://g' \
148             | sort | uniq -c | sort -nr | head -50
149     fi
150 }

153 #########################################
154 # 5. SUSPICIOUS PATTERN EVALUATOR
155 #########################################
156 
157 detect_suspicious_activity() {
158     local pcap="$1"
159     [[ -f "$pcap" ]] || { echo "No pcap file available."; return; }
160 
161     echo "===== Suspicious Activity (simple heuristic) ====="
162 
163     if [[ "$CAPTURE_TOOL" == "tshark" ]]; then
164         tshark -r "$pcap" -T fields \
165             -e ip.src -e ip.dst -e tcp.dstport \
166             -Y "tcp" 2>/dev/null \
167             | awk 'NF==3 {print $1, $2, $3}' \
168             | sort | uniq \
169             | awk '{key=$1"->"$2; ports[key]=ports[key]","$3}
170             END {for (k in ports) {n=split(ports[k],a,","); if (n > '"$MIN_SUSPICIOUS_PORTS"') print k" ports="n}}' \
171         | sort -k2 -nr
172 else
173     tcpdump -nn -r "$pcap" tcp 2>/dev/null \
174         | awk '{print $3, $5}' | sed 's/://g' \
175         | sort | uniq \
176         | awk '{key=$1"->"$2; c[key]++}
177         END {for (k in c) if (c[k] > '"$MIN_SUSPICIOUS_PORTS"') print k" connections="c[k]}' \
178     | sort -k2 -nr
179     fi
180 }

183 #####################################
184 # 6. REPORT BUILDER
185 #####################################
186 
187 build_report () {
188     local pcap="$1"
189     local out_dir="$2"
190 
191     [[ -f "$pcap" ]] || { log_warn "Cannot build report: pcap missing."; return; }
192 
193     local ts
194     ts="$(timestamp)"
195     local report="${out_dir}/network_traffic_report_${ts}.txt"
196 
197     set +e
198     set +u
199     {
200         echo "==============================================="
201         echo " Network Traffic Analyzer (Local) - Report"
202         echo " Version: ${VERSION}"
203         echo " Date:    $(date)"
204         echo " File:    ${pcap}"
205         echo "==============================================="
206         echo
207         echo "1. Protocol Summary"
208         echo "-------------------"
209         analyze_protocols "$pcap"
210         echo
211         echo "2. IP Endpoints"
212         echo "--------------"
213         analyze_endpoints "$pcap"
214         echo
215         echo "3. TCP Connections"
216         echo "-------------------"
217         analyze_connections "$pcap"
218         echo
219         echo "4. Suspicious Activity"
220         echo "----------------------"
221         detect_suspicious_activity "$pcap"
222         echo
223         echo "5. Notes"
224         echo "--------"
225         echo "- This report was generated from real captured traffic."
226         echo "- For best results, generate traffic from Ubuntu to Kali (ping, ssh, curl, nmap)."
227         echo "- Execute this script in your private home network, not in public Wi-Fi."
228         echo
229 } > "$report" 2>&1
230 set -e
231 set -u
232 
233 log_info "Report generated: $report"
234 }

236 #############################################
237 # 7. ORCHESTRATOR (Single & Watch Mode)
238 #############################################
239 
240 run_single_capture_and_report() {
241     local iface="$1"
242     local duration="$2"
243     local out_dir="$3"
244 
245     echo -e "${BLUE}[STATUS] Initializing capture directory...${RESET}"
246     init_capture_dir "$out_dir"
247 
248     echo -e "${BLUE}[STATUS] Detecting capture tool...${RESET}"
249     detect_capture_tool
250 
251     local pcap
252 
253     echo -e "${BLUE}[STATUS] Starting packet capture...${RESET}"
254     pcap="$(run_capture_once "$iface" "$duration" "$out_dir")"
255 
256     echo -e "${BLUE}[STATUS] Analyzing captured traffic...${RESET}"
257     echo -e "${BLUE}[STATUS] Generating report...${RESET}"
258     [[ -f "$pcap" ]] && build_report "$pcap" "$out_dir"
259 
260     echo -e "${BLUE}[STATUS] Process completed.${RESET}"
261 }
262 
263 
264 run_watch_mode() {
265     local iface="$1"
266     local duration="$2"
267     local interval="$3"
268     local out_dir="$4"
269 
270     init_capture_dir "$out_dir"
271     detect_capture_tool
272 
273     echo -e "${BLUE}[STATUS] Watch Mode started...${RESET}"
274     log_info "Watch Mode started."
275 
276     while true; do
277         local pcap
278         echo -e "${BLUE}[STATUS] Starting continuous capture iteration...${RESET}"
279         pcap="$(run_capture_once "$iface" "$duration" "$out_dir")"
280         echo -e "${BLUE}[STATUS] Analyzing captured traffic (continuous mode)...${RESET}"
281         echo -e "${BLUE}[STATUS] Generating report (continuous mode)...${RESET}"
282 
283         [[ -f "$pcap" ]] && build_report "$pcap" "$out_dir"
284 
285         echo -e "${BLUE}[STATUS] Waiting ${interval}s before next iteration...${RESET}"
286         log_info "Waiting ${interval}s before next iteration..."
287         sleep "$interval"
288     done
289 }

295 print_usage() {
296     cat <<EOF
297     Network Traffic Analyzer (Local) - v${VERSION}
298 
299     Usage:
300     $(basename "$0") [-i interface] [-d duration] [-o directory] [--watch] [--interval seconds]
301     Options:
302     -i IFACE        Network interface (default: ${DEFAULT_IFACE})
303     -d SECONDS      Capture duration (default: ${DEFAULT_CAPTURE_DURATION})
304     -o DIR          Output directory (default: ${DEFAULT_CAPTURE_DIR})
305     --watch         Enable Watch Mode
306     --interval N    Seconds between iterations (default: ${DEFAULT_WATCH_INTERVAL})
307     -h, --help      Show this help message
308 EOF
309 }
310 
311 main() {
312     echo -e "${BLUE}[STATUS] Starting Network Traffic Analyzer...${RESET}"
313 
314 local iface="${DEFAULT_IFACE}"
315 local duration="${DEFAULT_CAPTURE_DURATION}"
316 local out_dir="${DEFAULT_CAPTURE_DIR}"
317 local watch="false"
318 local interval="${DEFAULT_WATCH_INTERVAL}"
319 
320 
321 while [[ $# -gt 0 ]]; do
322 case "$1" in
323     -i) iface="$2"; shift 2 ;;
324     -d) duration="$2"; shift 2 ;;
325     -o) out_dir="$2"; shift 2 ;;
326     --watch) watch="true"; shift ;;
327     --interval) interval="$2"; shift 2 ;;
328     -h|--help) print_usage; exit 0 ;;
329     *) log_error "Unknown option: $1"; exit 1 ;;
330 esac
331 done
332 
333 if [[ "$watch" == "true" ]]; then
334 run_watch_mode "$iface" "$duration" "$interval" "$out_dir"
335 else
336 run_single_capture_and_report "$iface" "$duration" "$out_dir"
337 fi
338 }
339 
340 main "$@"
